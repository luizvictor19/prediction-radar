import 'dotenv/config';
import cron from 'node-cron';
import { collectAll } from './collectors/polymarket.js';
import { runAllDetectors } from './detectors/runner.js';
import { runRetentionJob } from './jobs/retention.js';

async function main(): Promise<void> {
  console.log('[main] Prediction Radar starting...');

  try {
    await collectAll();
  } catch (err) {
    console.error('[main] Initial collection failed (will retry on cron):', err);
  }

  // Collector: every 3 minutes
  cron.schedule('*/3 * * * *', () => { void collectAll(); });

  // Detector runner: every 5 minutes
  cron.schedule('*/5 * * * *', () => { void runAllDetectors(); });

  // Retention job: daily at 3am UTC
  cron.schedule('0 3 * * *', () => { void runRetentionJob(); });

  console.log('[main] Cron jobs scheduled. Running.');
}

main().catch((err) => {
  console.error('[main] Fatal error:', err);
  process.exit(1);
});

// trackConversation(conversation, ctx, signalIdWithOutcome)
// signalIdWithOutcome formats:
//   "uuid"         → cross_market (outcome from signal.suggested_outcome)
//   "uuid:outcome" → calendar_driven ou outros (outcome forçado, nome real do evento)
//
// Roteamento:
//   cross_market_inter → basketTrack() — coleta stake total, distribui entre legs
//   outros             → singleLegTrack() — fluxo single-leg
//
// singleLegTrack flow: stake → TO WIN (vem do campo "to win" do Polymarket) → confiança (skip) → tese (skip) → confirm
//   entry_price calculado = stake / to_win
//   shares = to_win (= quanto recebe se ganhar)
//   On confirm: inserts my_bets + my_bet_legs, marks signal acted_on=true, adjustCash(-stakeUsd)
//
// basketTrack flow (cross_market_inter):
//   Step 1: stake total
//   Step 2: exibe execução (N legs: outcome, stake por leg, shares, entry_price)
//   Step 3: tese (skip) → confirm
//   On confirm: inserts my_bets + N my_bet_legs, adjustCash(-totalStake)
//

NOTAS DE DESIGN — DECISÕES RECENTES
================================================================================

1. Bankroll = cash + portfolio (não mais campo único)
   - system_config.cash_usd: dinheiro disponível (não investido)
   - Portfolio: calculado em tempo real via mid_price das legs abertas
   - bankroll = cash + portfolio — exibido em /status e /topup, /withdraw
   - /topup e /withdraw substituem o antigo /bankroll
   - Track e close position chamam adjustCash() para manter cash sincronizado

2. Modelo de dados: my_bets + my_bet_legs (schema normalizado)
   - my_bets: cabeçalho (thesis, category, placed_at, closed_at, signal_id)
   - my_bet_legs: cada leg individual com entry_price, stake, shares, result, pnl
   - Basket = 1 my_bets + N my_bet_legs com mesmo bet_id
   - Single-leg = 1 my_bets + 1 my_bet_leg

3. Input "to win" em vez de preço decimal
   - /track e /register (single-leg) pedem "valor a receber se ganhar" (vem do Polymarket)
   - entry_price calculado = stake_usd / to_win
   - shares = to_win (porque 1 share paga $1 se ganhar)
   - /edit campo "To win" recalcula entry_price e shares
   - Basket legs: usam entry_price decimal diretamente (não usa to_win)

4. Outcomes reais (não hard-coded Yes/No)
   - normalizeOutcome(): match case-insensitive contra events.outcomes.values
   - /register e /edit (campo outcome) normalizam via normalizeOutcome()
   - track_yes/track_no callbacks resolvem outcomes[0]/outcomes[1] do evento
   - calendarDrivenKeyboard recebe outcomes do evento para botões dinâmicos
   - Emojis dinâmicos: ✅/❌ Yes/No, ⬆️/⬇️ Over/Under, 🪙/🎲 empate, 👑/🐺 favorito

5. Cap de stake por signal_type
   - cross_market_inter / cross_market_intra → config.cross_market_max_stake_pct (default 10%)
   - outros (calendar_driven, hype_reality_gap, …) → config.max_stake_pct (default 3%)
   - helper getStakeCap(config, signalType) em format.ts

6. calendar_driven não tem expected_edge_pct na metadata
   - Queries que filtram por expected_edge_pct usam .or() para incluir calendar_driven:
     .or(`signal_type.eq.calendar_driven,metadata->>expected_edge_pct.gte.${threshold}`)
   - suggested_outcome = null; outcome vem do callback (track_yes / track_no)
   - track.ts parseia signalIdWithOutcome como "uuid" ou "uuid:outcome_name"

7. Callbacks registrados no bot:
   - track:{id}        → cross_market (outcome vem de signal.suggested_outcome)
   - track_yes:{id}    → outcome = events.outcomes.values[0]
   - track_no:{id}     → outcome = events.outcomes.values[1]
   - dismiss:{id}
   - close:{bet_id}    → fechar posição (single ou basket)
   - edit_leg:{leg_id} → editar leg específica

8. URLs Polymarket:
   - cross_market: lookup via members[0].polymarket_id → event_group_slug
   - calendar_driven: lookup via event_id → event_group_slug

9. Collector: single-pass Gamma API (sem CLOB)
   - Insere snapshot para AMBOS os outcomes por evento (ex: Yes e No)
   - negRiskMarketID com baixo volume é incluído para viabilizar basket arb
   - polymarket_category vem de market.feeType (canônico da Polymarket)
   - is_ai_tech = categorizeMarket().category !== 'other' (flag interna, não filtra coleta)
   - sports_market_type e line mapeados de GammaMarket → Event

10. Event match em /register
    - Usa RPC trigram search_events_by_title (similarity_threshold=0.15)
    - Retorna até N matches com botões de escolha
    - Se match encontrado: herda category; outlook procura sinal ativo para linkar

    - MAX_CHARS = 3800 (limite Telegram é 4096, margem de segurança)
    - Chunks 2+ são threads reply ao chunk 1
