import 'dotenv/config';
import cron from 'node-cron';
import { collectAll } from './collectors/polymarket.js';
import { collectOpenLegMarkets } from './collectors/open-legs-collector.js';
import { collectEarlyMarkets } from './collectors/early-markets-collector.js';
import { runAllDetectors } from './detectors/runner.js';
import { runRetentionJob } from './jobs/retention.js';

async function main(): Promise<void> {
  // Collector geral: a cada 3 min (chama detectResolvedMarkets no final)
  cron.schedule('*/3 * * * *', () => { void collectAll(); });

  // Open legs: a cada 10 segundos (preços frescos)
  cron.schedule('*/10 * * * * *', () => { void collectOpenLegMarkets(); });

  // Detector runner: a cada 15 min (lock anti-overlap interno)
  cron.schedule('*/15 * * * *', () => { void runAllDetectors(); });

  // Early markets: a cada 5 min
  cron.schedule('*/5 * * * *', () => { void collectEarlyMarkets(); });

  // Retention: diário às 3h UTC + roda no startup
  cron.schedule('0 3 * * *', () => { void runRetentionJob(); });
  void runRetentionJob();

  // Fire-and-forget iniciais
  void collectOpenLegMarkets();
  void collectEarlyMarkets();
  void collectAll();
}

// detectResolvedMarkets(seenPolymarketIds: Set<string>): chamado no final de collectAll()
// seenPolymarketIds = ids de markets com closed=false vistos no ciclo atual
//
// Candidate selection em 2 etapas:
//   Etapa 1: events com leg aberta (PRIORITÁRIOS — buscados sem limit)
//   Etapa 2: events ativos com end_date 90d atrás até 30d no futuro (limit 500, ord end_date asc)
//   Merge: prioritários + normais (sem duplicar); MAX_PER_CYCLE=50 para os normais
//
// Missing = candidates cujo polymarket_id NÃO está em seenPolymarketIds
// Para cada missing: GET /markets/{polymarket_id} (timeout 8s)
//   - market.closed=false → log partial e ignorar (liquidez/pausa?)
//   - market.closed=true mas umaResolutionStatus ≠ 'resolved' → skip (pending UMA)
//   - outcomePrices=['0.5','0.5'] → void (anulado)
//   - outcomePrices com '1' em uma posição → winner found
//
// closeLegsForResolvedEvent():
//   - win: resolution_price=1.0, pnl = (1 - entry_price) × shares, adjustCash(shares)
//   - loss: resolution_price=0.0, pnl = -stake_usd (sem adjustCash)
//   - void: pnl=0, adjustCash(stake_usd) — devolve o stake
//   - Calcula CLV = último snapshot mid_price antes de resolved_at
//   - Fecha my_bets quando todas as legs estiverem closed
// Componente no log: 'resolved_detector'

NOTAS DE DESIGN — DECISÕES RECENTES
================================================================================

1. Bankroll = cash + portfolio (campo separado)
   - /cash substitui /topup e /withdraw — define valor absoluto, não incremento
   - Track e close position chamam adjustCash() para manter cash sincronizado
   - adjustCash(+shares) ao fechar win; adjustCash(-stake) ao abrir

2. Modelo de dados: my_bets + my_bet_legs
   - Single-leg = 1 my_bets + 1 my_bet_leg
   - Basket = 1 my_bets + N my_bet_legs
   - "To win" em vez de preço decimal: entry_price = stake / to_win; shares = to_win

3. Detectores ativos (6):
   - cross_market_intra, cross_market_inter, calendar_driven
   - hype_reality_gap (implementado, FILTRO 15/85 — só mercados em zona de movimento)
   - early_market (novo, FILTRO 5/95 — exclui cauda extrema)
   - cleanup_stale_signals (job de manutenção)
   - dismissStaleSignals() no runner — dismiss automático por last_seen_at stale

4. Coletores ativos (4):
   - polymarket.ts (3min) — principal, detecta resolvidos no final
   - open-legs-collector.ts (10s) — preços frescos das legs abertas
   - early-markets-collector.ts (5min) — mercados < 24h com is_new_market=true
   - retention.ts (daily 3am + startup)

5. Auto-resolver (resolved-detector.ts)
   - Roda ao final de cada collectAll()
   - 2 etapas: events com leg aberta (sem limit) + outros (limit 500)
   - Só processa se market sumiu do feed (não está em seenPolymarketIds)
   - Valida umaResolutionStatus='resolved' antes de fechar
   - Fecha legs, calcula CLV, ajusta cash automaticamente

6. Filtros de cauda nos detectores:
   - hype_reality_gap: mid_price < 0.15 ou > 0.85 → skip
   - early_market: currentPrice < 0.05 ou > 0.95 → skip

7. TTL e cooldowns via system_config (configuráveis sem deploy):
   - signal_ttl_minutes (default 30) — TTL de sinais
   - signal_cooldown_minutes (default 60) — cooldown após dismiss
   - dismiss_stale_cutoff_minutes (default 15) — runner dismissStaleSignals cutoff
   - stale_cleanup_threshold_hours (default 1) — cleanup job threshold

8. Callbacks novos no bot:
   - track_custom:{id}        → customTrackConversation (tese própria para cross_market_inter)
   - analyze_ai:{id}          → stub ("em breve") — buildSignalContext() pronto para Fase 5
   - close_leg_select:{betId} → mostra lista de legs com botões individuais
   - close_single_leg:{legId} → fecha 1 leg específica de basket

9. /signals melhorias:
   - Oculta sinais de events com leg aberta (já no portfólio)
   - Suporta filtro de texto: /signals bitcoin
   - hype_reality_gap e early_market passam sem edge mínimo
   - Order: calendar_driven (por end_date) primeiro, depois por edge desc

10. Keyboard layout por signal_type:
    - cross_market_inter: Track basket / ✏️ Tese própria / 🧠 Analisar / Dismiss
    - calendar_driven: Track YES / Track NO / 🧠 Analisar / Dismiss
    - hype_reality_gap / early_market: Track Out0 / Track Out1 / 🧠 Analisar / Dismiss
    - outros: Track / Dismiss + 🧠 Analisar

11. Retention job
    - Usa snapshot_retention_days do banco (não hardcoded)
    - Roda daily às 3h UTC + no startup
