import type { SystemConfig } from '../types/index.js';
import { supabase } from './supabase.js';
import { logEvent } from './logger.js';

const CACHE_TTL_MS = 60 * 1000;

const DEFAULTS: SystemConfig = {
  id: 1,
  bankroll_usd: 500,
  max_stake_pct: 0.03,
  cross_market_max_stake_pct: 0.10,
  kelly_fraction: 0.25,
  min_confidence_alert: 0.70,
  drawdown_stop_pct: 0.20,
  telegram_chat_id: null,
  daily_report_hour: 9,
  cross_market_log_threshold: 0.03,
  cross_market_high_confidence_threshold: 0.08,
  cross_market_dedup_window_minutes: 30,
  inter_market_min_members: 3,
  inter_market_min_total_volume_24h: 10000,
  snapshot_retention_days: 7,
  system_logs_retention_days: 30,
  min_expected_edge_pct: 1.5,
  notify_min_edge_pct: 2.0,
  log_expected_edge_pct: 0.5,
  excluded_categories: [],
  collector_min_volume_24h: 10000,
  collector_min_liquidity: 20000,
  updated_at: new Date().toISOString(),
};

let cachedConfig: SystemConfig | null = null;
let cacheExpiresAt = 0;

export async function getSystemConfig(): Promise<SystemConfig> {
  const now = Date.now();
  if (cachedConfig !== null && now < cacheExpiresAt) return cachedConfig;

  const { data, error } = await supabase
    .from('system_config')
    .select('*')
    .eq('id', 1)
    .single();

  if (error || !data) {
    await logEvent({
      component: 'config',
      status: 'error',
      message: `Failed to load system_config, using defaults: ${error?.message ?? 'no data'}`,
    });
    return DEFAULTS;
  }

  cachedConfig = data as SystemConfig;
  cacheExpiresAt = now + CACHE_TTL_MS;
  return cachedConfig;
}

export function invalidateConfigCache(): void {
  cachedConfig = null;
  cacheExpiresAt = 0;
}

import 'dotenv/config';
import cron from 'node-cron';
import { collectAll } from './collectors/polymarket.js';
import { runAllDetectors } from './detectors/runner.js';
import { runRetentionJob } from './jobs/retention.js';

async function main(): Promise<void> {
  console.log('[main] Prediction Radar starting...');

  try {
    await collectAll();
  } catch (err) {
    console.error('[main] Initial collection failed (will retry on cron):', err);
  }

  // Collector: every 3 minutes
  cron.schedule('*/3 * * * *', () => { void collectAll(); });

  // Detector runner: every 5 minutes
  cron.schedule('*/5 * * * *', () => { void runAllDetectors(); });

  // Retention job: daily at 3am UTC
  cron.schedule('0 3 * * *', () => { void runRetentionJob(); });

  console.log('[main] Cron jobs scheduled. Running.');
}

main().catch((err) => {
  console.error('[main] Fatal error:', err);
  process.exit(1);

NOTAS DE DESIGN — DECISÕES RECENTES
================================================================================

1. Cap de stake por signal_type
   - cross_market_inter / cross_market_intra → config.cross_market_max_stake_pct (default 10%)
   - outros (calendar_driven, hype_reality_gap, …) → config.max_stake_pct (default 3%)
   - helper getStakeCap(config, signalType) em format.ts

2. calendar_driven não tem expected_edge_pct na metadata
   - Queries que filtram por expected_edge_pct usam .or() para incluir calendar_driven:
     .or(`signal_type.eq.calendar_driven,metadata->>expected_edge_pct.gte.${threshold}`)
   - suggested_outcome = null; outcome vem do callback (track_yes / track_no)
   - track.ts parseia signalIdWithOutcome como "uuid" ou "uuid:yes"/"uuid:no"

3. Callbacks registrados no bot:
   - track:{id}      → cross_market (outcome vem de signal.suggested_outcome)
   - track_yes:{id}  → calendar_driven YES
   - track_no:{id}   → calendar_driven NO
   - dismiss:{id}
   - close:{id}      → fechar posição

4. URLs Polymarket:
   - cross_market: lookup via members[0].polymarket_id → event_group_slug
   - calendar_driven: lookup via event_id → event_group_slug

5. Collector: single-pass Gamma API (sem CLOB)
   - negRiskMarketID com baixo volume é incluído para viabilizar basket arb
   - polymarket_category vem de market.feeType (canônico da Polymarket)
   - is_ai_tech = categorizeMarket().category !== 'other' (flag interna, não filtra coleta)

6. Categorizer (src/collectors/categorizer.ts)
   - Filtra apenas por title (não description) — evita falsos positivos de URLs/boilerplate
   - Exclusões: Claude Giroux (NHL), Meta/Russia geopolítico, Apple com case-sensitive
   - isRelevantMarket() usada para logging; 'other' ainda é coletado (para basket arb)

7. message-utils.ts: mensagens longas no Telegram
   - MAX_CHARS = 3800 (limite Telegram é 4096, margem de segurança)
   - Chunks 2+ são threads reply ao chunk 1

8. Detector runner: cross_market_intra, cross_market_inter, calendar_driven (ativos)
   hype_reality_gap comentado (stub)

9. src/telegram/ vs src/bot/
   - src/telegram/ contém stubs para roadmap futuro (Week 5-6)
   - src/bot/ é a implementação real e atual
